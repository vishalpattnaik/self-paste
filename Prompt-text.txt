import org.apache.maven.plugin.AbstractMojo;
import org.apache.maven.plugins.annotations.Mojo;
import org.apache.maven.plugins.annotations.Parameter;
import org.springframework.boot.builder.SpringApplicationBuilder;
import org.springframework.context.ConfigurableApplicationContext;
import java.io.File;

@Mojo(name = "upgrade", threadSafe = true)
public class UpgradeMojo extends AbstractMojo {

    @Parameter(defaultValue = "${project.basedir}", readonly = true, required = true)
    private File projectBaseDir;

    // This captures the exact recipe the developer wants to run
    @Parameter(property = "recipe", required = true)
    private String recipeName;

    @Override
    public void execute() {
        getLog().info("Starting intentional tech-debt upgrade...");
        getLog().info("Target Recipe: " + recipeName);

        try (ConfigurableApplicationContext context = new SpringApplicationBuilder(CveScannerApplication.class)
                .logStartupInfo(false)
                .run()) {
            
            // Bypass the Orchestrator and CVE DB completely!
            // Just call your existing execution service directly.
            OpenRewriteExecutionService rewriteService = context.getBean(OpenRewriteExecutionService.class);
            
            boolean success = rewriteService.executeRecipe(projectBaseDir.getAbsolutePath(), recipeName);
            
            if (success) {
                getLog().info("Upgrade complete. Please review the changes and run 'mvn clean test'.");
            } else {
                getLog().info("No changes were necessary for this upgrade.");
            }

        } catch (Exception e) {
            getLog().error("Upgrade failed: " + e.getMessage());
        }
    }
}
